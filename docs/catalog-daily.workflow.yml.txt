name: catalog-daily
# Fetches Edmunds' rich product feed from Impact once a day and refreshes the catalogue database.
# Secrets (repository settings): IMPACT_FTP_USER, IMPACT_FTP_PASS, DATABASE_URL_DIRECT (direct, non-pooled connection string).
on:
  schedule:
    - cron: "30 8 * * *" # 08:30 UTC, about three hours after Impact's update (05:33 UTC on Oct 12, 2026)
  workflow_dispatch:
    inputs:
      sample_every:
        description: "Test mode for a small database: load only every Nth car (for example 5). Leave empty for the full load."
        required: false
        default: ""
concurrency:
  group: catalog-daily
  cancel-in-progress: false
jobs:
  load:
    runs-on: ubuntu-latest
    timeout-minutes: 90
    steps:
      - uses: actions/checkout@v4
        with:
          ref: catalog-db/v1 # scheduled workflows run from the DEFAULT branch (main), so the loader code is checked out explicitly
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: npm
      - run: npm ci
      - name: Download the rich feed (encrypted SFTP first, pinned host key; plain FTP only as a fallback)
        env:
          IMPACT_FTP_USER: ${{ secrets.IMPACT_FTP_USER }}
          IMPACT_FTP_PASS: ${{ secrets.IMPACT_FTP_PASS }}
        run: |
          set -uo pipefail
          # Host key fingerprint read from a screenshot on Oct 12, 2026: verify it before relying on it.
          HOSTKEY="VnpT96wJuWk5n+mOCab3Tw084xZgZ6ojm4ZtRh82vTs"
          if curl --fail -sS --retry 3 --hostpubsha256 "$HOSTKEY" -u "$IMPACT_FTP_USER:$IMPACT_FTP_PASS" -o feed.txt.gz "sftp://products.impact.com/Edmunds/Edmunds-Product-Feed_CUSTOM.txt.gz"; then
            echo "downloaded over SFTP"
          else
            echo "SFTP failed; falling back to plain FTP (the server offers no FTPS)"
            curl --fail -sS --retry 3 --ftp-pasv -u "$IMPACT_FTP_USER:$IMPACT_FTP_PASS" -o feed.txt.gz "ftp://products.impact.com/Edmunds/Edmunds-Product-Feed_CUSTOM.txt.gz"
          fi
          ls -l feed.txt.gz
          test "$(stat -c %s feed.txt.gz)" -gt 50000000
          gzip -t feed.txt.gz
      - name: Load into the database (aborts without changes if the feed is truncated)
        env:
          DATABASE_URL_DIRECT: ${{ secrets.DATABASE_URL_DIRECT }}
          SAMPLE_EVERY: ${{ github.event.inputs.sample_every }}
        run: |
          if [ -n "$SAMPLE_EVERY" ]; then
            npx tsx scripts/catalog-load.ts feed.txt.gz --sample-every "$SAMPLE_EVERY"
          else
            npx tsx scripts/catalog-load.ts feed.txt.gz --expect-rows 1300000
          fi
