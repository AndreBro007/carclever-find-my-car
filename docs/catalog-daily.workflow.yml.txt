name: catalog-daily
# Fetches Edmunds' rich product feed from Impact and refreshes the catalogue database.
# All logic lives in the app repository (branch catalog-db/v1: scripts/fetch-feed.sh and scripts/load-feed.sh),
# so it can change without editing this file. Secrets (settings of THIS repository):
# IMPACT_FTP_USER, IMPACT_FTP_PASS, DATABASE_URL_DIRECT (direct, non-pooled connection string).
on:
  # schedule is OFF until the Neon upgrade (a full load needs about 1.1 GB, over the free plan). To switch on, remove the two # below.
  # schedule:
  #   - cron: "30 8 * * *" # 08:30 UTC, about three hours after Impact's update (05:33 UTC on Oct 12, 2026)
  workflow_dispatch:
    inputs:
      sample_every:
        description: "Test mode for a small database: load only every Nth car (for example 5). Leave empty for the full load."
        required: false
        default: ""
concurrency:
  group: catalog-daily
  cancel-in-progress: false
jobs:
  load:
    runs-on: ubuntu-24.04
    timeout-minutes: 90
    steps:
      - uses: actions/checkout@v7
        with:
          repository: AndreBro007/carclever-find-my-car # this file lives in the docs repository; the code is checked out from the app repository
          ref: catalog-db/v1
      - uses: actions/setup-node@v7
        with:
          node-version: 22
          cache: npm
      - run: npm ci
      - name: Download and verify the rich feed
        env:
          IMPACT_FTP_USER: ${{ secrets.IMPACT_FTP_USER }}
          IMPACT_FTP_PASS: ${{ secrets.IMPACT_FTP_PASS }}
        run: bash scripts/fetch-feed.sh feed.txt.gz
      - name: Load into the database
        env:
          DATABASE_URL_DIRECT: ${{ secrets.DATABASE_URL_DIRECT }}
          SAMPLE_EVERY: ${{ github.event.inputs.sample_every }}
        run: bash scripts/load-feed.sh feed.txt.gz
